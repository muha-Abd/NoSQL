CREATE KEYSPACE IF NOT EXISTS library_lab
WITH replication = {
    'class': 'SimpleStrategy',
    'replication_factor': 1
};

USE library_lab;

CREATE TABLE IF NOT EXISTS loans_by_reader (
    reader_id text,
    loan_date date,
    loan_time time,
    loan_id text,
    book_id text,
    book_title text,
    category text,
    due_date date,
    return_date date,
    status text,
    PRIMARY KEY ((reader_id), loan_date, loan_time, loan_id)
);
